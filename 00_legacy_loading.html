Cleaning confirmaciones

Need to

  1. Normalize empty strings to NA
  2. Identify and resolve duplicate weekly records
  3. Validate cleaned confirmations
  4. Transform them into two PostgreSQL tables

1. Normalize empty strings to NA

# Normalize empty strings
confirmaciones_completas <- old_data$confirm |>
  # select(id, volId, dias) |> 
  unnest(cols = dias) |>
  clean_names() |> 
  pivot_longer(c(lunes, martes, miercoles, jueves, viernes), names_to = "dia", values_to = "estado") |>
  mutate(estado = na_if(estado, ""),
         centro = na_if(centro, ""),
         obs = na_if(obs, ""))

# Check duplicates
confirm_dupes <- confirmaciones_completas |> 
  count(vol_id, semana, bloque, dia) |> 
  filter(n > 1)

# Compare full values for each duplicate record
dupe_compare <- confirmaciones_completas |> 
  semi_join(confirm_dupes, by = c("vol_id", "semana", "bloque", "dia")) |> 
  arrange(semana, bloque, dia, id)

4. Transform into two tables for postgres

Add temp IDs to weekly confirmations

confirmacion_semanal <- confirmacion_semanal |> 
  mutate(id = row_number(), .before = 1)

Add confirmation id’s to daily confirmations

confirmacion_dia <- confirmacion_dia |> 
  left_join(confirmacion_semanal |> 
              select(id, vol_id, semana, bloque),
            by = c("vol_id", "semana", "bloque")) |> 
  rename(confirmacion_id = id) |> 
  select(confirmacion_id, dia, estado)

confirmacion_dia |>
  summarise(n_registros = n(),
            ids_faltantes = sum(is.na(confirmacion_id)),
